SELECT d, COUNT(*) races, SUM(pub IS NOT NULL) published, SUM(pub IS NULL) not_published,
 ROUND(AVG(TIMESTAMPDIFF(SECOND,r2s,r2r))) r2_avg_sec, MAX(TIMESTAMPDIFF(SECOND,r2s,r2r)) r2_max_sec,

FROM (SELECT LEFT(s.race_id,8) d, s.race_id, s.frozen_at frz, s.sales_close_at close_at,
