── config/queue.php に足した1節（全文）──────────────────────
        // ══════════════════════════════════════════════════════════════
        // @ANCHOR-20261021-R122-QUEUECONF  r12.2 専用の待ち行列（010の2章）
        // ══════════════════════════════════════════════════════════════
        //   既定の QUEUE_CONNECTION は sync のままです。
        //   この接続は app/Jobs/R122*Job が名指しで使うだけなので、
        //   アプリの他の場所の動きは変わりません。
        //
        //   retry_after は worker の上限より十分長くします。
        //     RunRoleJob の上限  175秒
        //     worker の --timeout 180秒
        //     retry_after        300秒
        //   短いと、まだ動いているJobを別のworkerが拾って二重に動きます。
        //
        //   after_commit は true です。トランザクションの中で dispatch
        //   しても、commit まで実際には積まれません。未commitの
        //   snapshot を worker が先に読むことを防ぎます（010の2-2）。
        'r122' => [
            'driver' => 'database',
            'table' => 'jobs',
            'queue' => 'r122-agg',
            'retry_after' => 300,
            'after_commit' => true,
        ],

── 待ち行列の表（DDL。php artisan migrate は打ちません）──────
  このプロジェクトには database/migrations に baganriki 系の
  migration が1本もありません。r12.2 の表も DDL で作っています。
  migrate を打つと、関係のない未適用の migration まで走るので
  打ちません。定義は Laravel 8 の queue:table ／
  queue:batches-table ／ queue:failed-table が作るものと同じ形です。

    jobs         (id, queue, payload, attempts, reserved_at,
                  available_at, created_at, KEY jobs_queue_index)
    job_batches  (id, name, total_jobs, pending_jobs, failed_jobs,
                  failed_job_ids, options, cancelled_at, created_at,
                  finished_at)
    failed_jobs  (id, uuid UNIQUE, connection, queue, payload,
                  exception, failed_at)
