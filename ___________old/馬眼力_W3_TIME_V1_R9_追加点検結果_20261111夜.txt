def all_paths(value, path=()):
    yield path, value
    if isinstance(value, dict):
        for key, child in value.items():
            yield from all_paths(child, path + (key,))
    elif isinstance(value, list):
        for index, child in enumerate(value):
            yield from all_paths(child, path + (index,))


def get(value, path):
    for key in path: value = value[key]
    return value


def reexpress(value):
    if isinstance(value, dict):
        return {key: reexpress(child) for key, child in reversed(list(value.items()))}
    if isinstance(value, list): return [reexpress(child) for child in value]
    if type(value) is int: return Decimal(str(value)+'.000')
    return value
